    <script>
        let pagAtual = 1;
        const totalPags = 2;

        function mudarPagina(dir) {
            let nova = pagAtual + dir;
            if (nova >= 1 && nova <= totalPags) {
                document.getElementById('p' + pagAtual).classList.remove('ativa');
                document.getElementById('p' + nova).classList.add('ativa');
                pagAtual = nova;
                atualizarProgresso();
                document.querySelector('.livro-container').scrollTop = 0;
            }
        }

        function atualizarProgresso() {
            let perc = Math.round((pagAtual / totalPags) * 100);
            document.getElementById('bar-fill').style.width = perc + '%';
            document.getElementById('label-progresso').innerText = `Você concluiu ${perc}% da leitura deste capítulo`;
        }

        function mudarTema(t) {
            if (t === 'noturno') document.body.classList.add('tema-noturno');
            else document.body.classList.remove('tema-noturno');
        }

        atualizarProgresso();
    </script>
